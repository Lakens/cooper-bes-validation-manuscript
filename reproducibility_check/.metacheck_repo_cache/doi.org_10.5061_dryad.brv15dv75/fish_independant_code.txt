#t test for size comparison (time in open rivers)
Data <- read.csv("Size-Nov-Feb-Open.csv")

#test for normality
hist(Data$width[Data$time=="November"])
hist(Data$width[Data$time=="February"])
tapply(Data$width,Data$time,shapiro.test)

#if not normal, transform with SQRT
Data$widthSQ <- sqrt(Data$width)
hist(Data$widthSQ[Data$time=="November"])
hist(Data$widthSQ[Data$time=="February"])
tapply(Data$widthSQ,Data$time,shapiro.test)

#if not normal, transform with log10
Data$widthLG <- log10(Data$width)
hist(Data$widthLG[Data$time=="November"])
hist(Data$widthLG[Data$time=="February"])
tapply(Data$widthLG,Data$time,shapiro.test)

#test for homogeneity of variance
boxplot(Data$width~Data$time)
leveneTest(Data$width~Data$time)

#t test
t.test(Data$width~Data$time, mu=0, alt="two.sided", conf=0.95, var.eq=T, paired=F)

#if not normal
res <- wilcox.test(width~time, data = Data,exact = FALSE)
res
